## Katana

# Download Linux Binary:
https://github.com/projectdiscovery/katana/releases/tag/v1.7.0
./katana -u https://freshforensics.tech -d 3 -silent -o urls.txt
cat urls.txt
sort -u urls.txt -o urls.txt
python Python/EyeWitness.py -f urls.txt -d eyewitness-report --headless

## wget (web-crawler)
wget \
  --spider \
  --recursive \
  --no-verbose \
  --level=3 \
  https://freshforensics.tech 2>&1 |
grep '^--' |
awk '{print $3}' |
sort -u
