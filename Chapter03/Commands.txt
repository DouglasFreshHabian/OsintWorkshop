# Eyewitness
git clone https://github.com/RedSiege/EyeWitness.git
cd Eyewitness/setup
sudo ./setup.sh
cd ..
source eyewitness-venv/bin/activate
python Python/EyeWitness.py --single https://example.com
eyewitness --single https://freshforensics.tech -d freshforensics-screens
nano urls.txt
eyewitness -f urls.txt -d freshforensics-screens

# Katana
https://github.com/projectdiscovery/katana
katana -u https://freshforensics.tech -d 3 -silent -o urls.txt
cat urls.txt
sort -u urls.txt -o urls.txt
eyewitness -f urls.txt -d eyewitness-report --headless

# wget (web-crawler)
wget \
  --spider \
  --recursive \
  --no-verbose \
  --level=3 \
  https://freshforensics.tech 2>&1 |
grep '^--' |
awk '{print $3}' |
sort -u
